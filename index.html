const PROJECT_URL = Deno.env.get("SUPABASE_URL")!; const SERVICE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!; const searches = [ { q: "Turkmenistan", lang: "en", hl: "en-US", gl: "US", ceid: "US:en" }, { q: "Türkmenistan", lang: "tk", hl: "tr", gl: "TR", ceid: "TR:tr" }, { q: "Türkmenistan", lang: "tr", hl: "tr", gl: "TR", ceid: "TR:tr" }, { q: "Туркменистан", lang: "ru", hl: "ru", gl: "RU", ceid: "RU:ru" } ]; function decodeXml(value: string) { return value.replace(/&/g, "&") .replace(/</g, "<") .replace(/>/g, ">") .replace(/"/g, '"') .replace(/'/g, "'"); } function getTag(xml: string, tag: string) { const match = xml.match( new RegExp(`<${tag}(?:\\s[^>]*)?>([\\s\\S]*?)<\\/${tag}>`, "i") ); return match ? decodeXml(match[1].replace(//g, "$1").trim()) : ""; } async function getFeed(search: typeof searches[number]) { const url = "https://news.google.com/rss/search?q=" + encodeURIComponent(search.q) + "&hl=" + encodeURIComponent(search.hl) + "&gl=" + search.gl + "&ceid=" + encodeURIComponent(search.ceid); const response = await fetch(url); if (!response.ok) throw new Error("Google News HTTP " + response.status); const xml = await response.text(); const items = xml.match(/]*)?>[\s\S]*?<\/item>/gi) || []; return items.map(item => { const title = getTag(item, "title").replace(/<[^>]*>/g, "").trim(); const link = getTag(item, "link"); const pubDate = getTag(item, "pubDate"); const source = getTag(item, "source").replace(/<[^>]*>/g, "").trim(); return { title, canonical_url: link, source_name: source || "Google News", language: search.lang, published_at: pubDate && !Number.isNaN(Date.parse(pubDate)) ? new Date(pubDate).toISOString() : null, summary: "Google News RSS üzerinden bulunan haber. Tam içerik için kaynak bağlantısını açın.", status: "published" }; }).filter(item => item.title && item.canonical_url && /^https?:\/\//i.test(item.canonical_url) ); } Deno.serve(async (request) => { if (request.method === "OPTIONS") { return new Response("ok", { headers: { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, apikey, content-type" } }); } if (!PROJECT_URL || !SERVICE_KEY) { return Response.json( { error: "SUPABASE_URL veya SUPABASE_SERVICE_ROLE_KEY eksik." }, { status: 500 } ); } try { const feeds = await Promise.allSettled(searches.map(getFeed)); const unique = new Map(); for (const feed of feeds) { if (feed.status !== "fulfilled") continue; for (const item of feed.value) { if (!unique.has(item.canonical_url)) { unique.set(item.canonical_url, item); } } } const rows = [...unique.values()].slice(0, 100); if (!rows.length) { return Response.json({ ok: true, fetched: 0, message: "Bu çalıştırmada haber sonucu alınamadı." }); } const result = await fetch( PROJECT_URL + "/rest/v1/tgn_news?on_conflict=canonical_url", { method: "POST", headers: { "apikey": SERVICE_KEY, "Authorization": "Bearer " + SERVICE_KEY, "Content-Type": "application/json", "Prefer": "resolution=ignore-duplicates,return=minimal" }, body: JSON.stringify(rows) } ); if (!result.ok) { return Response.json({ error: "Supabase kayıt hatası", details: await result.text() }, { status: 500 }); } return Response.json({ ok: true, fetched: rows.length, message: "Google News sonuçları Supabase'e gönderildi." }); } catch (error) { return Response.json({ error: String(error) }, { status: 500 }); } });
